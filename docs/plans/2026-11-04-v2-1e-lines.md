> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or

**Goal:**
- **Lines block:** a `lines` block shows the FinLite ledger lines behind a cube cell, ranked by supplier, project or
  account: top N, Other, without <group>, lines total, cube value and gap. Every number is a live SUMIFS on table
  `Lines_Q<n>_<alias>`, checked "Sheet = model".
- **`fin_drill`:** a shortcut that asks one question with just a lines block.
- **Clean refusals:** a refusal reads as a plain input error, never as "Unexpected … run sv_restart".
- **Shorter formulas:** SUMIFS criteria that are constant for the whole Data table are left out.

**Architecture:**
- **Model.**
  - A `LineSet` per lines block lives on the `Frame` (`frame.lines[alias]`).
  - The expression `LinePick` sums `AMOUNT` over the line set's rows. `CubeRef` is the cube value of the drilled
    cell. Both evaluate in Python and render to Excel like every other tree.
  - The builder `blocks/lines.py` is pure.
- **Drill.** It runs in the async layer of `fin_analyze`, before `run_prepare`, through v1's
  `core/drillstep.drill`. Its rows reach the model through `prepare(lines=...)`.
- **Writing.** The Excel job writes each line set with v1's `drill_xl.write_lines` (table `Lines_Q<n>_<alias>`,
  reused in place on a redraw) and a "Cube value" cell. Then the answer is drawn.

**Facts from v1:**
- **Drill entry:** `async drillstep.drill(cell, *, say, connection="FinLite", max_lines, split) -> DrillResult`, with
  `rows: list[dict]`, `columns`, `cell_value`, `reconciliation`, `warnings`. The cell must be Periodic. It holds
  one account, one entity and one period member, which may be a parent such as `Q3 26` or `Yr 26`.
- **Line columns:** `AMOUNT` (float), `AP_CUSTOMER_NAME` (supplier), `ACCOUNT`, `PROJECT` / `PROJECT_DESC`,

### Task 2: Shorter SUMIFS: leave out constant criteria

**Files:** `engine/smartass/model/expr.py`, `engine/smartass/render/layout.py`. Test:
`engine/tests/v2/test_v2_short_formulas.py`.

**Rule:** a filter whose single value equals the one value a column has in the whole Data table matches every row,
so it is left out of the formula. This is exactly equivalent. A follow-up or refresh redraws every formula, so a
later wider table is always covered.

```python
# engine/smartass/model/lines.py
"""Ledger lines behind one cube cell (a FinLite drill), as one set per lines block. Written to Excel as table
Lines_Q<n>_<alias> on sheet 'Q<n> Lines <alias>' (v1 drill_xl.write_lines), with the cube value in D2."""
from __future__ import annotations

from dataclasses import dataclass, field

AMOUNT = "AMOUNT"
CUBE_CELL = "$D$2"


@dataclass
class LineSet:
    alias: str
    rows: list[dict]
    columns: list[str]
    cube_value: float | None = None
    reconciliation: dict = field(default_factory=dict)
    warnings: list[str] = field(default_factory=list)


def norm(v) -> str:
    return str(v or "").strip().casefold()
```

  In `expr.py`:

```python
@dataclass(frozen=True)
class LinePick(Expr):
    """Sum of AMOUNT over a line set's rows whose `column` (case-insensitive, trimmed) is one of `values`
    (column None: every row). Excel: SUMIFS on table Lines_Q<n>_<alias> (n from the Data table name)."""
    alias: str
    column: str | None
    values: tuple[str, ...] | None

    def evaluate(self, frame, cells):
        from .lines import AMOUNT, norm
        ls = frame.lines[self.alias]
        want = None if self.column is None else {norm(v) for v in self.values or ()}
        hit = [float(r.get(AMOUNT) or 0.0) for r in ls.rows
               if want is None or norm(r.get(self.column)) in want]
        return sum(hit) if hit else None

    def to_excel(self, table, addr, const=None):
        t = f"Lines_{table.split('_', 1)[1]}_{self.alias}"
        if self.column is None:
            return f"SUM({t}[AMOUNT])"
        vals = tuple(dict.fromkeys(self.values or ()))
        if len(vals) == 1:
            return f"SUMIFS({t}[AMOUNT],{t}[{self.column}],{_crit(vals[0])})"
        arr = "{" + ",".join(_crit(v) for v in vals) + "}"
        return f"SUM(SUMIFS({t}[AMOUNT],{t}[{self.column}],{arr}))"


@dataclass(frozen=True)
class CubeRef(Expr):
    """The cube value of the drilled cell, written next to the lines (sheet 'Q<n> Lines <alias>', D2)."""
    alias: str

    def evaluate(self, frame, cells):
        return frame.lines[self.alias].cube_value

    def to_excel(self, table, addr, const=None):
        from .lines import CUBE_CELL
        q = table.split("_", 1)[1]                      # 'Q3'
        return f"'{q} Lines {self.alias}'!{CUBE_CELL}"
```

  Use the existing criterion quoting helper. It is `_crit` after the 1A fixes; it quotes and escapes `~*?`.
  Because SUMIFS is case-insensitive, list only distinct spellings. The empty string must render as `""`, which
  matches blank cells.

```python
"""lines: the ledger lines behind one cube cell, grouped by supplier (AP_CUSTOMER_NAME), project (INITIATIVE) or
account (ACCOUNT): top N (biggest absolute amount first; spellings that differ only in case are one group), Other,
lines without a <group>, the lines total, the cube value and the gap (cube - lines), share of the lines total."""
from __future__ import annotations

from ..expr import CubeRef, LinePick, Ref, Sub, Sum, share
from ..lines import AMOUNT, norm
from ..spec import LinesBlock
from .common import BlockOut, Line, ck

COLUMN = {"supplier": "AP_CUSTOMER_NAME", "project": "INITIATIVE", "account": "ACCOUNT"}
NOUN = {"supplier": "supplier", "project": "booked project", "account": "account"}


def build_lines(b: LinesBlock, ctx, key: str) -> BlockOut:
    ls = ctx.frame.lines[key]
    col = COLUMN[b.group_by]
    groups: dict[str, list] = {}
    for r in ls.rows:
        name = str(r.get(col) or "").strip()
        if name:
            g = groups.setdefault(norm(name), [name, 0.0, set()])
            g[1] += float(r.get(AMOUNT) or 0.0)
            g[2].add(name)
    ranked = sorted(groups.values(), key=lambda g: -abs(g[1]))
    shown, rest = ranked[:b.top], ranked[b.top:]
    out = BlockOut(key, "lines", b.title or f"Ledger lines by {b.group_by}", ["Amount", "Share %"])
    out.cols = ["amount", "share"]
    for i, (name, _, spellings) in enumerate(shown):
        out.lines.append(Line(f"g{i}", name, [LinePick(key, col, tuple(sorted(spellings))),
                                              share(Ref(ck(key, f"g{i}", 0)), Ref(ck(key, "total", 0)))]))
    named = Sum(tuple(Ref(ck(key, f"g{i}", 0)) for i in range(len(shown)))) if shown else None
    if rest:
        other = Sub(Sub(Ref(ck(key, "total", 0)), Ref(ck(key, "blank", 0))), named)
        out.lines.append(Line("other", "Other", [other, share(Ref(ck(key, "other", 0)), Ref(ck(key, "total", 0)))]))
    out.lines.append(Line("blank", f"Lines without a {NOUN[b.group_by]}",
                          [LinePick(key, col, ("",)), share(Ref(ck(key, "blank", 0)), Ref(ck(key, "total", 0)))]))
    out.lines.append(Line("total", "Ledger lines total", [LinePick(key, None, None), None], kind="sum"))
    out.lines.append(Line("cube", "Cube value", [CubeRef(key), None]))
    out.lines.append(Line("gap", "Gap (cube - lines)", [Sub(Ref(ck(key, "cube", 0)), Ref(ck(key, "total", 0))), None]))
    return out
```
