# Speed step 1: drills without Excel and without a page per piece - Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or
> superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A drill (sv_drill, lines_ranking) never touches Excel and loads at most one drill page per browser session;
every piece is two direct requests from that warm tab.

**Architecture:** The Smart View provider over HTTP (`svhttp`, already used for grids) also answers the drill link
(`GetDrillThroughReports`) and the split's child values. A new `drill_direct.DirectDrill` keeps one warm drill tab
in the shared drill browser and, per piece, GETs EPM's own drill query (`&FORMAT=SUI`) and pages `getData` with it.
`core/drillstep.py` uses both first and falls back (logged) to today's routes: Smart View in Excel for links and
values, the tab pump (a page per piece) for lines.

**Tech Stack:** Python 3.12, Playwright (sync, on the drill browser's worker thread), pytest, uv.

**Spec:** `docs/specs/2026-11-02-speed-design.md` (step 1). **Evidence:** `docs/audit/2026-11-02-drill-speed-spike.md`.

```python
def drill_reports(sid: str, cube: str, ords: int, cells: list[tuple[str, str, str]]) -> str:
    """List the drill-through reports of ONE data cell (read-only; never ExecuteDrillThroughReport). Captured from
    Smart View 25.100 (2026-11-02): `ords` is the cell's row-major index in the grid just retrieved; `cells` is
    (dimension, 'row' | 'col' | 'pov', Default-alias label) for every dimension, in grid order."""
    dims = "".join(
        f'<dim name="{html.escape(d, quote=True)}" '
        f'{"pov=\"pov\"" if where == "pov" else f"{where}=\"0\""} enc="0">{_x(label)}</dim>'
        for d, where, label in cells)
    return (f"<req_GetDrillThroughReports><sID>{_x(sid)}</sID><usr></usr><pwd></pwd><rowGov></rowGov>"
            f"<timeGov></timeGov><cube>{_x(cube)}</cube><ords>{int(ords)}</ords><alsTbl>Default</alsTbl><pov/>"
            f"<cells>{dims}</cells></req_GetDrillThroughReports>")


def drill_url(text: str) -> tuple[str, str] | None:
    """(report name, URL relative to the EPM base without './') of the first drill report, or None."""
    raise_for_exception(text)
    m = re.search(r'<drillThroughURL\b([^>]*)>(.*?)</drillThroughURL>', text, re.S)
    if not m or not m.group(2).strip():
        return None
    name = re.search(r'\bname="([^"]*)"', m.group(1))
    return (html.unescape(name.group(1)) if name else "", html.unescape(m.group(2)).strip().lstrip("./"))
```

```python
    def drill_link(self, conn: str, cell: dict[str, str]) -> dict:
        """ops.drill_link over HTTP: {cube_value, url (relative to the EPM base), report}. One 1x1 retrieve gives
        the value and the Default-alias labels Smart View sends; then GetDrillThroughReports (read-only)."""
        from .ops import DRILL_DEFAULTS, drill_cell_defaults
        cell, _ = drill_cell_defaults(cell)
        if cell.get("All Balance Types", "Periodic") != "Periodic":
            raise InputError("Drill-through works on Periodic data only; use a period member (quarter/year) "
                             "instead of Accumulated.")
        pov = {**DRILL_DEFAULTS, **{d: m for d, m in cell.items() if d not in ("All Accounts", "All Time Periods")}}
        f = self.fetch(conn, pov, [Axis("All Accounts", [cell["All Accounts"]])],
                       [Axis("All Time Periods", [cell["All Time Periods"]])])
        lay, v = f.layout, f.values
        cells = [("All Time Periods", "col", str(v[1][lay.nr])), ("All Accounts", "row", str(v[1 + lay.nc][0]))]
        cells += [(d, "pov", str(v[0][lay.nr + j])) for j, (d, _) in enumerate(lay.pov)]
        ords = (1 + lay.nc) * len(v[0]) + lay.nr
        val = v[1 + lay.nc][lay.nr]
        if not self._lock.acquire(timeout=config.SVHTTP_TIMEOUT_S):
            raise SvHttpError("the HTTP route is busy with another query")
        try:
            got = self._ask(conn, lambda sid, cube: svxml.drill_reports(sid, cube, ords, cells), svxml.drill_url,
                            lambda e: False)
        finally:
            self._lock.release()
        return {"cube_value": val if isinstance(val, float) else None,
                "url": got[1] if got else None, "report": got[0] if got else None}
```

Note: `_ask`'s `read` returns whatever `svxml.drill_url` returns (a tuple or None); `_ask` is typed `-> list` but
returns `read(ans)` unchanged - change its annotation to `-> object` while here.

```python
    def child_values(self, conn: str, parts: list[dict]) -> list | None:
        """drill_split.child_values_on over HTTP: the parts differ in one dimension - one column retrieve,
        float or None per part (None for the whole list when they differ in nothing)."""
        from .ops import DRILL_DEFAULTS
        first = parts[0]
        dim = next((k for k in first if any(p.get(k) != first[k] for p in parts)), None)
        if dim is None:
            return None
        col = "All Time Periods" if dim != "All Time Periods" else "All Accounts"
        pov = {**DRILL_DEFAULTS, **{k: v for k, v in first.items() if k not in (dim, col)}}
        f = self.fetch(conn, pov, [Axis(dim, [p[dim] for p in parts])], [Axis(col, [first[col]])])
        lay = f.layout
        return [v if isinstance(v, float) else None
                for v in (f.values[1 + lay.nc + i][lay.nr] for i in range(len(parts)))]
```

```python
"""Drill lines without loading a drill page per piece (spike 2026-11-02, docs/audit/2026-11-02-drill-speed-spike.md).

Per piece, from ONE warm tab (a tab that loaded one real drill page): GET the drill URL's JSON form
(&FORMAT=SUI) - EPM's own drill query - and page getData with it, field for field ({QUERY, PARAMS: PARAMSARRAY,
PARAMSTYPE: PARAMSTYPEARRAY}). Read-only: nothing is built by us. getData from a tab that never loaded a drill page
gets the whole EPM session signed out, so it is only ever sent from the warm tab."""
from __future__ import annotations

import json
from dataclasses import dataclass
from urllib.parse import parse_qsl, quote, urlencode, urlsplit, urlunsplit

from .drill_browser import CellTooWide, DrillError


class NoDrillData(DrillError):
    """EPM's drill query has no lines here (data not loaded through Data Integration, or an empty cell)."""


class SignedOut(DrillError):
    """EPM answered with something that is not JSON (its sign-in / exit page)."""


@dataclass
class SuiQuery:
    body: str               # the getData POST body
    app_id: int | None
    partition: int | None


def sui_url(base: str, url: str, now_ms: int) -> str:
    """The drill URL's JSON form: DRILLURL dropped, FORMAT=SUI and a cache buster added (as the page asks it)."""
    u = urlsplit(base.rstrip("/") + "/" + url.lstrip("./"))
    q = [(k, v) for k, v in parse_qsl(u.query, keep_blank_values=True) if k != "DRILLURL"]
    q += [("FORMAT", "SUI"), ("_", str(now_ms))]
    return urlunsplit((u.scheme, u.netloc, u.path, urlencode(q, quote_via=quote, safe="."), u.fragment))


def parse_sui(text: str) -> SuiQuery:
    try:
        j = json.loads(text)
    except ValueError:
        raise SignedOut("EPM answered the drill query with a page, not JSON (signed out?)") from None
    if not isinstance(j, dict):
        raise SignedOut("EPM's drill query answer is not a JSON object")
    if "QUERY" not in j:
        why = str(j.get("details") or "no drill query in EPM's answer")
        if "EPMFDM-140373" in why:
            raise CellTooWide(why)
        if "does not return any data" in why:
            raise NoDrillData(why)
        raise DrillError(f"EPM refused the drill: {why}")
    body = json.dumps({"QUERY": j["QUERY"], "PARAMS": j.get("PARAMSARRAY", []),
                       "PARAMSTYPE": j.get("PARAMSTYPEARRAY", [])})
    return SuiQuery(body, j.get("APPLICATION_ID"), j.get("PARTITIONKEY"))
```

Check: `urlencode(..., safe=".")` must leave `attribute=Entities.id.un.SY` and `sso_token=T` untouched; the test
pins the exact string. If the token contains characters `quote` changes, compare with the spike: the page sends the
URL as Smart View gave it.

### Task 5: DirectDrill - one warm tab, two requests per piece

**Files:**
- Modify: `smartview_mcp/drill_direct.py` (append the class)
- Test: `tests/test_drill_direct.py`

Behaviour (all on the drill browser's worker thread, via `DrillBrowser.submit`):
1. `b._ensure(s)` (start + sign in if needed).
2. SUI GET from the warm tab if there is one, else from `s.page` (SUI is safe from any signed-in tab). Headers:
   `X-Requested-With: XMLHttpRequest`, `Accept: */*`, `Content-Type: application/json`, plus the warm tab's
   `page_id` / `x-appbuilder-client-id` when known.
3. `parse_sui` -> `NoDrillData` / `CellTooWide` propagate as they are.
4. No warm tab (or it was closed) -> warm: `s.ctx.new_page()`, `goto` this piece's URL, wait (poll 200 ms, up to
   `config.DRILL_CAPTURE_TIMEOUT_S`) for the page's first `getData?limit=` request; keep the tab and that request's
   headers (minus cookie/content-length/host). No request in time -> `DrillError("warm drill page never asked for
   its data")`.
5. View definition: GET `/aif/ui/model/view?page=DRILL_THROUGH&appId=<app_id>&partitionkey=<partition>` from the
   warm tab, cached per `(app_id, partition)`.
6. Lines: `drill_browser.page_through(post, max_lines, progress, recover)` with
   `post(off)` = fetch POST `base + /aif/ui/model/drilldown/getData?limit=<DRILL_PAGE>&offset=<off>` from the warm
   tab with the warm headers and the SUI body; `recover()` = drop the warm tab, `b._login(s)`, warm again.
7. Return `{"items", "total", "view", "secs"}` - the tab pump's contract.

```python
import concurrent.futures as cf
import logging
import threading
import time
from typing import Callable

from . import config
from .drill_browser import page_through

log = logging.getLogger("smartview.drill")
_GET = """async ([u, h]) => { try { const r = await fetch(u, {headers: h, credentials: 'include'});
                               return {s: r.status, t: await r.text()}; } catch (e) { return {s: 0, t: String(e)}; } }"""
_POST = """async ([u, b, h]) => { try { const r = await fetch(u, {method: 'POST', body: b, headers: h,
                                   credentials: 'include'}); return {s: r.status, t: await r.text()}; }
                                   catch (e) { return {s: 0, t: String(e)}; } }"""
_SUI_HEADERS = {"X-Requested-With": "XMLHttpRequest", "Accept": "*/*", "Content-Type": "application/json"}
_DROP = ("cookie", "content-length", "host")


class DirectDrill:
    """Worker-thread state: the warm tab, its getData headers and the view cache. One per drill browser session;
    use fetch() from any thread, fetch_on() on the worker thread (tests call it directly)."""

    def __init__(self, capture_timeout_s: float | None = None):
        self.tab = None
        self.headers: dict[str, str] = {}
        self.views: dict[tuple, dict] = {}
        self.capture_timeout_s = config.DRILL_CAPTURE_TIMEOUT_S if capture_timeout_s is None else capture_timeout_s

    # ---------------------------------------------------------------- worker thread
    def _warm(self, b, s, url: str) -> None:
        p = s.ctx.new_page()
        got: dict = {}

        def on_request(req):
            if "/aif/ui/model/drilldown/getData?limit=" in req.url and "h" not in got:
                got["h"] = {k: v for k, v in dict(req.headers).items() if k.lower() not in _DROP}
        p.on("request", on_request)
        try:
            p.goto(b.base + "/" + url.lstrip("./"), wait_until="domcontentloaded", timeout=60_000)
            t0 = time.monotonic()
            while "h" not in got and time.monotonic() - t0 < self.capture_timeout_s:
                p.wait_for_timeout(200)
        finally:
            p.remove_listener("request", on_request)
        if "h" not in got:
            p.close()
            raise DrillError("The warm drill page never asked for its data; the page route takes over.")
        self.tab, self.headers = p, got["h"]
        log.info("direct drill: warm tab ready")

    def _drop_tab(self) -> None:
        if self.tab is not None:
            try:
                self.tab.close()
            except Exception:   # noqa: BLE001 - already closed
                pass
        self.tab, self.headers, self.views = None, {}, {}

    def _alive(self) -> bool:
        return self.tab is not None and not self.tab.is_closed()

    def _sui_headers(self) -> dict:
        extra = {k: v for k, v in self.headers.items() if k.lower() in ("page_id", "x-appbuilder-client-id")}
        return {**_SUI_HEADERS, **extra}

    def fetch_on(self, b, s, url: str, max_lines: int,
                 progress: Callable[[int, int | None], None] | None = None) -> dict:
        b._ensure(s)
        tab = self.tab if self._alive() else s.page
        r = tab.evaluate(_GET, [sui_url(b.base, url, int(time.time() * 1000)), self._sui_headers()])
        q = parse_sui(r["t"])                          # NoDrillData / CellTooWide / SignedOut / DrillError
        if not self._alive():
            self._warm(b, s, url)
        key = (q.app_id, q.partition)
        if key not in self.views:
            v = self.tab.evaluate(_GET, [f"{b.base}/aif/ui/model/view?page=DRILL_THROUGH&appId={q.app_id}"
                                         f"&partitionkey={q.partition}&_={int(time.time() * 1000)}",
                                         self._sui_headers()])
            try:
                self.views[key] = json.loads(v["t"])
            except ValueError:
                raise SignedOut("EPM answered the drill view with a page, not JSON (signed out?)") from None

        def post(off: int) -> tuple[int, str]:
            res = self.tab.evaluate(_POST, [f"{b.base}/aif/ui/model/drilldown/getData?limit={config.DRILL_PAGE}"
                                            f"&offset={off}", q.body, self.headers])
            return int(res["s"]), res["t"]

        def recover() -> None:
            self._drop_tab()
            b._login(s)
            self._warm(b, s, url)
        t1 = time.monotonic()
        items, total = page_through(post, max_lines, progress, recover)
        return {"items": items, "total": total, "view": self.views.get(key), "secs": round(time.monotonic() - t1, 1)}

    # ---------------------------------------------------------------- any thread
    def fetch(self, url: str, max_lines: int) -> cf.Future:
        from .drill_browser import shared_browser
        b = shared_browser()
        return b.submit(lambda s: self.fetch_on(b, s, url, max_lines))


_shared: DirectDrill | None = None
_shared_lock = threading.Lock()


def shared_direct() -> DirectDrill:
    global _shared
    with _shared_lock:
        if _shared is None:
            _shared = DirectDrill()
        return _shared
```
