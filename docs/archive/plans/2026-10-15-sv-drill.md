# sv_drill Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A read-only MCP tool `sv_drill` that drills any FinLite cell to its ledger lines (supplier, invoice, voucher, line text) through a hidden, headless browser. It reconciles the lines to the cube, writes them as an Excel table in the session workbook, and lets `sv_ws_answer` build adaptive reports on top with `SUMIFS`.

**Architecture:**
1. Smart View in the server's hidden Excel retrieves the cell and returns its Data Integration drill URL (new bridge function `B_DrillReports`).
2. A persistent headless Google Chrome (Playwright, `channel="chrome"`) signs in through Telenor SSO and opens the drill page. It captures the page's own `getData` request and replays it with `offset` to page through all lines.
3. Pure modules normalise the lines (types, Norwegian text repair), reconcile them, profile them, plan splits and store them in SQLite.
4. The tool writes a live FinLite grid and a `Linjer_Q<n>` table into the user's session workbook. The answer spec gets a new cell kind `{"drill": ...}`.

**Tech Stack:** Python 3.14 (venv `%LOCALAPPDATA%\svkit\venv`), MCP SDK 2.3 (`MCPServer`), pywin32 COM, Oracle Smart View VBA (`HsAddin`), Playwright for Python (sync API, Google Chrome channel), sqlite3, pytest.

```python
"""Pure logic for drill-through lines (no Excel, no browser): the drill page's columns, typed values, repair of
Norwegian text that EPM delivers broken, reconciliation to the cube, and a profile Claude uses to pick a report.

Lines are dicts keyed by the drill page's column LABEL ("AMOUNT", "AP_CUSTOMER_NAME", ...), in page order."""
from __future__ import annotations

import collections
import datetime as dt
import json
import re
from pathlib import Path

AMOUNT = "AMOUNT"
NUM_FIELDS = {"AMOUNTX", "DESC1", "DESC2", "ATTR10", "ATTR30"}      # amount, debit, credit, trans/orig amount
DATE_FIELDS = {"ATTR4", "ATTR17"}                                    # booking date, GL posted date
KEY_LABELS = ["Period", "GL_POSTED_DATE", "BOOKING_DATE", "GL_VOUCHER_NUMBER", "Entity", "CC_DESC", "ACCOUNT",
              "ACCOUNT_NAME", "AMOUNT", "JE_SOURCE", "JE_CATEGORY_NAME", "AP_CUSTOMER_NAME", "AP_INVOICE_NUMBER",
              "GL_DESCRIPTION_LINE", "Products", "PRODUCT_DESC", "PROJECT", "PROJECT_DESC", "BATCH_NAME",
              "JOURNAL_NAME"]
FACETS = {"supplier": "AP_CUSTOMER_NAME", "account": "ACCOUNT", "cost_centre": "Entity", "month": "Period",
          "je_source": "JE_SOURCE", "je_category": "JE_CATEGORY_NAME", "product": "Products",
          "invoice": "AP_INVOICE_NUMBER"}
FACET_DESC = {"account": "ACCOUNT_NAME", "cost_centre": "CC_DESC", "product": "PRODUCT_DESC",
              "invoice": "AP_CUSTOMER_NAME"}
_MOJIBAKE = re.compile("[\u00c3\u00c2][\u0080-\u00bf]")              # 'Ã¥' = UTF-8 'å' read as Latin-1


def columns_from_view(view: dict, name: str = "DrillFilter") -> list[tuple[str, str]]:
    """(label, field) pairs from /aif/ui/model/view?page=DRILL_THROUGH (the page's default view)."""
    items = view.get("items") or []
    v = next((x for x in items if x.get("viewName") == name), items[0] if items else None)
    if v is None:
        raise ValueError("The drill view definition has no views.")
    return [(c["label"], c["value"]) for c in json.loads(v["viewDefinition"])]


def load_text_fixes(path: Path) -> dict[str, str]:
    try:
        return dict(json.loads(Path(path).read_text(encoding="utf-8")).get("fixes", {}))
    except (OSError, ValueError):
        return {}


def repair_text(s: str, fixes: dict[str, str], stats: dict) -> str:
    """Undo double-encoded UTF-8 exactly; replace fragments EPM lost (U+FFFD) from the correction list; report
    whatever is still broken. Good text is returned unchanged."""
    if _MOJIBAKE.search(s):
        try:
            s = s.encode("latin-1").decode("utf-8")
            stats["fixed_mojibake"] += 1
        except (UnicodeEncodeError, UnicodeDecodeError):
            pass
    if "\ufffd" in s:
        for bad, good in fixes.items():
            s = s.replace(bad, good)
        if "\ufffd" in s:
            stats["unfixed"].add(s)
        else:
            stats["fixed_by_list"] += 1
    return s


def _value(field: str, v):
    if v is None or v == "":
        return None
    if field in NUM_FIELDS:
        try:
            return float(v)
        except (TypeError, ValueError):
            return str(v)
    if field in DATE_FIELDS:
        try:
            return dt.datetime.fromisoformat(str(v)[:19])
        except ValueError:
            return str(v)
    return v if isinstance(v, str) else str(v)


def normalize(items: list[dict], columns: list[tuple[str, str]], fixes: dict[str, str]) -> tuple[list[dict], dict]:
    """Raw getData items -> rows keyed by label, typed, text repaired. Returns (rows, text_repair_stats)."""
    stats = {"fixed_mojibake": 0, "fixed_by_list": 0, "unfixed": set()}
    rows = []
    for x in items:
        r = {}
        for label, field in columns:
            v = _value(field, x.get(field))
            r[label] = repair_text(v, fixes, stats) if isinstance(v, str) else v
        rows.append(r)
    stats["unfixed"] = sorted(stats["unfixed"])
    return rows, stats


def _amounts(rows: list[dict]) -> list[float]:
    return [r[AMOUNT] for r in rows if isinstance(r.get(AMOUNT), float)]


def reconcile(cube_value, rows: list[dict]) -> dict:
    """Cube cell vs sum of lines. debit = sum of negative amounts, credit = sum of positive (FinMax signs)."""
    a = _amounts(rows)
    s = round(sum(a), 2)
    cube = None if cube_value is None else round(float(cube_value), 2)
    gap = None if cube is None else round(cube - s, 2)
    status = "no cube value" if gap is None else ("exact" if abs(gap) < 0.005 else "gap")
    return {"cube_value": cube, "lines_sum": s, "gap": gap, "status": status, "lines": len(rows),
            "debit": round(sum(x for x in a if x < 0), 2), "credit": round(sum(x for x in a if x > 0), 2)}


def _facet(rows: list[dict], label: str, top: int, desc: str | None = None) -> dict:
    acc: dict = collections.defaultdict(lambda: [0.0, 0, None])
    for r in rows:
        e = acc[r.get(label) or ""]
        e[0] += r[AMOUNT] if isinstance(r.get(AMOUNT), float) else 0.0
        e[1] += 1
        if desc and not e[2]:
            e[2] = r.get(desc)
    ordered = sorted(acc.items(), key=lambda kv: -abs(kv[1][0]))
    out = []
    for k, (amount, n, d) in ordered[:top]:
        g = {"key": k, "amount": round(amount, 2), "lines": n}
        if desc:
            g["desc"] = d
        out.append(g)
    return {"groups": len(acc), "top": out}


def profile(rows: list[dict], top: int = 10) -> dict:
    """What Claude needs to choose report sections: top groups per facet by |amount|, lines without supplier and
    the journals behind them (allocations, accruals), whether signs are mixed (then no 'share of total')."""
    p = {name: _facet(rows, label, top, FACET_DESC.get(name)) for name, label in FACETS.items()}
    no_sup = [r for r in rows if not r.get("AP_CUSTOMER_NAME")]
    p["no_supplier"] = {"lines": len(no_sup), "amount": round(sum(_amounts(no_sup)), 2)}
    p["journals_without_supplier"] = _facet(no_sup, "JOURNAL_NAME", top)
    untagged = [r for r in rows if r.get("Products") in (None, "pd.No_Product")]
    p["untagged_product"] = {"lines": len(untagged), "amount": round(sum(_amounts(untagged)), 2)}
    a = _amounts(rows)
    p["signs_mixed"] = any(x < 0 for x in a) and any(x > 0 for x in a)
    return p
```

```python
"""Split a drill cell into child cells when it holds too many lines (pure; children come from the atlas).
Order: period (year -> quarters -> months), then entity, then account. The caller recurses on pieces that are
still too big, so each call goes down one level of one dimension."""
from __future__ import annotations

from typing import Callable

ORDER = ("All Time Periods", "All Entities", "All Accounts")


def plan_split(cell: dict[str, str], children: Callable[[str, str], list[str]],
               order: tuple[str, ...] = ORDER) -> list[dict[str, str]] | None:
    """Child cells along the first dimension in `order` whose member has children; None if all are leaves."""
    for dim in order:
        member = cell.get(dim)
        kids = children(dim, member) if member else []
        if kids:
            return [{**cell, dim: k} for k in kids]
    return None


def atlas_children(connection: str) -> Callable[[str, str], list[str]]:
    """children(dim, member) from the local atlas (instant)."""
    from . import atlas
    db = atlas.connect()

    def kids(dim: str, member: str) -> list[str]:
        return [m["name"] for m in atlas.children(db, connection, member) if m.get("dimension", dim) == dim]
    return kids
```

```python
"""Local cache of drilled lines (%LOCALAPPDATA%\\svkit\\drill.db, outside OneDrive). Follow-up reports and key
points read the lines from here; drills older than DRILL_KEEP_DAYS are pruned."""
from __future__ import annotations

import datetime as dt
import json
import sqlite3
import uuid
from pathlib import Path

from . import config

_SCHEMA = """
CREATE TABLE IF NOT EXISTS drill (id TEXT PRIMARY KEY, created TEXT, connection TEXT, cell TEXT, rec TEXT,
                                  columns TEXT, workbook TEXT, q INT, alias TEXT);
CREATE TABLE IF NOT EXISTS line (drill_id TEXT, n INT, data TEXT, PRIMARY KEY (drill_id, n));
CREATE INDEX IF NOT EXISTS drill_wq ON drill (workbook, q);
"""


def connect(path: Path = config.DRILL_DB) -> sqlite3.Connection:
    Path(path).parent.mkdir(parents=True, exist_ok=True)
    db = sqlite3.connect(path)
    db.executescript(_SCHEMA)
    return db


def _enc(v):
    return {"__dt": v.isoformat()} if isinstance(v, dt.datetime) else v


def _dec(d: dict) -> dict:
    return {k: dt.datetime.fromisoformat(v["__dt"]) if isinstance(v, dict) and "__dt" in v else v
            for k, v in d.items()}


def save(db, *, connection: str, cell: dict, rec: dict, columns: list[str], rows: list[dict], workbook: str | None,
         q: int | None, alias: str, created: dt.datetime | None = None) -> str:
    did = uuid.uuid4().hex[:12]
    with db:
        if workbook and q is not None:            # a re-drill of the same question/alias replaces the old one
            for (old,) in db.execute("SELECT id FROM drill WHERE workbook=? AND q=? AND alias=?",
                                     (workbook.lower(), q, alias)).fetchall():
                db.execute("DELETE FROM line WHERE drill_id=?", (old,))
                db.execute("DELETE FROM drill WHERE id=?", (old,))
        db.execute("INSERT INTO drill VALUES (?,?,?,?,?,?,?,?,?)",
                   (did, (created or dt.datetime.now()).isoformat(timespec="seconds"), connection,
                    json.dumps(cell), json.dumps(rec, default=str), json.dumps(columns),
                    (workbook or "").lower(), q, alias))
        db.executemany("INSERT INTO line VALUES (?,?,?)",
                       ((did, i, json.dumps({k: _enc(v) for k, v in r.items()}, ensure_ascii=False))
                        for i, r in enumerate(rows)))
    return did


def load(db, drill_id: str) -> tuple[dict, list[dict]]:
    m = db.execute("SELECT created, connection, cell, rec, columns, workbook, q, alias FROM drill WHERE id=?",
                   (drill_id,)).fetchone()
    if m is None:
        raise KeyError(drill_id)
    meta = {"id": drill_id, "created": m[0], "connection": m[1], "cell": json.loads(m[2]), "rec": json.loads(m[3]),
            "columns": json.loads(m[4]), "workbook": m[5], "q": m[6], "alias": m[7]}
    rows = [_dec(json.loads(d)) for (d,) in db.execute("SELECT data FROM line WHERE drill_id=? ORDER BY n",
                                                        (drill_id,))]
    return meta, rows


def find(db, workbook: str, q: int) -> dict[str, str]:
    """alias -> drill id for one question of one workbook."""
    return {a: i for i, a in db.execute("SELECT id, alias FROM drill WHERE workbook=? AND q=?",
                                        (workbook.lower(), q))}


def prune(db, keep_days: int = config.DRILL_KEEP_DAYS) -> int:
    cut = (dt.datetime.now() - dt.timedelta(days=keep_days)).isoformat(timespec="seconds")
    old = [i for (i,) in db.execute("SELECT id FROM drill WHERE created < ?", (cut,))]
    with db:
        for i in old:
            db.execute("DELETE FROM line WHERE drill_id=?", (i,))
            db.execute("DELETE FROM drill WHERE id=?", (i,))
    return len(old)
```

```python
"""sv_drill in the user's workbook: the lines table ('Q<n> Linjer', Excel table Linjer_Q<n>) and the formulas
answer cells use over it. Formula rules (tn-formelstil): SUMIFS by default; an exact-match FILTER form where SUMIFS
would coerce the key ('Jul-26' -> a date, digit strings over 15 characters -> a rounded number)."""
from __future__ import annotations

import re

_DATE_LIKE = re.compile(r"^\s*(\d{1,4}[-./]\d{1,2}([-./]\d{1,4})?|[A-Za-z]{3,9}[-./ ]\d{1,4}|\d{1,2}[-./ ][A-Za-z]{3,9}([-./ ]\d{1,4})?)\s*$")


def table_name(q: int, alias: str = "d") -> str:
    return f"Linjer_Q{q}" if alias == "d" else f"Linjer_Q{q}_{alias}"


def _lit(v) -> str:
    return '"' + str(v).replace('"', '""') + '"'


def safe_for_sumifs(v) -> bool:
    s = str(v)
    return not (_DATE_LIKE.match(s) or (s.isdigit() and len(s) > 15))


def drill_formula(table: str, where: dict[str, str], agg: str = "sum") -> str:
    """agg: sum | count | debit (sum of negatives) | credit (sum of positives)."""
    amt = f"{table}[AMOUNT]"
    sign = {"debit": '"<0"', "credit": '">0"'}.get(agg)
    if all(safe_for_sumifs(v) for v in where.values()):
        crit = "".join(f",{table}[{k}],{_lit(v)}" for k, v in where.items())
        if agg == "count":
            return f"=COUNTIFS({crit[1:]})" if where else f"=COUNT({amt})"
        if sign:
            return f"=SUMIFS({amt},{amt},{sign}{crit})"
        return f"=SUMIFS({amt}{crit})" if where else f"=SUM({amt})"
    conds = [f"{table}[{k}]={_lit(v)}" for k, v in where.items()]
    if sign:
        conds.append(f"{amt}{sign.strip(chr(34))}")
    test = conds[0] if len(conds) == 1 else "*".join(f"({c})" for c in conds)
    if agg == "count":
        return f"=SUM(--({test}))" if len(conds) == 1 else f"=SUM({test})"
    return f"=SUM(FILTER({amt},{test},0))"
```

```python
    need = {"All Accounts", "All Time Periods", "All Entities"}

                        "of Accumulated.")
    pov = {**DRILL_DEFAULTS, **{d: m for d, m in cell.items() if d not in ("All Accounts", "All Time Periods")}}
    lay, p, _ = query(h, conn, pov, [Axis("All Accounts", [cell["All Accounts"]])],

    sheet = ensure_grid_sheet(h, conn)

    i = next((k for k, n in enumerate(names) if "source" in str(n).lower()), 0)
```

```python
"""Headless Google Chrome that reads Data Integration drill pages (proven 15.10.2026, see the sv_drill spec).

Why this route: the drill link's sso_token alone does not pass the IDCS cloud gate, and Smart View's own execute
call opens the user's browser. Chrome with the IT-policy 'Windows Accounts' extension signs in silently after a
click on 'Telenor SSO'. The drill page renders only after /HyperionPlanning/ was opened in the same tab.

All Playwright calls run on ONE worker thread (the sync API is thread-bound). Never visible: headless=True always.
Our chrome.exe processes are recognised by the profile path in their command line; only those are ever killed."""
from __future__ import annotations

import concurrent.futures as cf
import json
import logging
import queue
import threading
import time
from typing import Callable

import psutil

from . import config

log = logging.getLogger("smartview.drill")


class DrillError(Exception):
    pass


class TooManyLines(DrillError):
    def __init__(self, total: int):
        super().__init__(f"{total} lines")
        self.total = total


class DrillBrowser:
    def __init__(self, base: str = config.EPM_BASE, profile=config.DRILL_PROFILE):
        self.base, self.profile = base.rstrip("/"), profile
        self.q: queue.Queue = queue.Queue()
        self.thread = threading.Thread(target=self._loop, name="drill-chrome", daemon=True)
        self.thread.start()
        self.pw = self.ctx = self.page = None
        self.ready = False

    # ------------------------------------------------------------ worker thread
    def _loop(self):
        while True:
            fn, fut = self.q.get()
            if fn is None:
                break
            if fut.set_running_or_notify_cancel():
                try:
                    fut.set_result(fn())
                except BaseException as e:   # noqa: BLE001 - forwarded to the caller
                    fut.set_exception(e)

    def submit(self, fn: Callable) -> cf.Future:
        fut: cf.Future = cf.Future()
        self.q.put((fn, fut))
        return fut

    # ------------------------------------------------------------ session
    def _start(self):
        from playwright.sync_api import sync_playwright
        self.profile.mkdir(parents=True, exist_ok=True)
        self.pw = sync_playwright().start()
        self.ctx = self.pw.chromium.launch_persistent_context(
            str(self.profile), channel="chrome", headless=True,
            ignore_default_args=["--disable-extensions"],             # the SSO extension must load
            args=["--disable-blink-features=AutomationControlled"], viewport={"width": 1600, "height": 1000})
        self.page = self.ctx.pages[0] if self.ctx.pages else self.ctx.new_page()

    def _login_preload(self):
        """Open the EPM home (signs in via Telenor SSO if needed); the drill page needs its session state."""
        p = self.page
        p.goto(self.base + "/HyperionPlanning/", wait_until="domcontentloaded", timeout=90_000)
        t0, clicked = time.monotonic(), False
        while time.monotonic() - t0 < config.DRILL_LOGIN_TIMEOUT_S:
            url = p.url
            if "/ui/v1/signin" in url and not clicked:
                b = p.get_by_role("button", name="Telenor SSO")
                if b.count():
                    b.first.click()
                    clicked = True
            if url.startswith(self.base + "/HyperionPlanning/") and time.monotonic() - t0 > 3:
                try:
                    p.wait_for_load_state("networkidle", timeout=30_000)
                except Exception:   # noqa: BLE001 - a busy home page is fine once we are on it
                    pass
                self.ready = True
                return
            p.wait_for_timeout(1000)
        raise DrillError(f"Telenor SSO did not complete headless within {config.DRILL_LOGIN_TIMEOUT_S}s "
                         f"(last page: {self.page.url[:120]}). Nothing was shown on screen.")

    def _ensure(self):
        if self.ctx is None:
            self._start()
        if not self.ready:
            self._login_preload()

    # ------------------------------------------------------------ one drill
    def _fetch(self, url: str, max_lines: int, progress: Callable[[int, int], None] | None) -> dict:
        self._ensure()
        p, got = self.page, {}

        def on_request(req):
            if "/aif/ui/model/drilldown/getData?limit=" in req.url and "req" not in got:
                got["req"] = {"url": req.url, "body": req.post_data, "headers": dict(req.headers)}

        def on_response(resp):
            if "/aif/ui/model/view?page=DRILL_THROUGH" in resp.url and "view" not in got:
                try:
                    got["view"] = resp.json()
                except Exception:   # noqa: BLE001
                    pass
        p.on("request", on_request)
        p.on("response", on_response)
        try:
            p.goto(self.base + "/" + url.lstrip("./"), wait_until="domcontentloaded", timeout=90_000)
            t0 = time.monotonic()
            while time.monotonic() - t0 < config.DRILL_CAPTURE_TIMEOUT_S and not ("req" in got and "view" in got):
                if "/ui/v1/signin" in p.url:                 # session expired: log in again, then retry once
                    self.ready = False
                    self._login_preload()
                    p.goto(self.base + "/" + url.lstrip("./"), wait_until="domcontentloaded", timeout=90_000)
                p.wait_for_timeout(500)
            if "req" not in got:
                raise DrillError("The drill page never asked for its data within "
                                 f"{config.DRILL_CAPTURE_TIMEOUT_S}s (Oracle UI change, or a blank page). "
                                 f"Page: {p.url[:120]}")
        finally:
            p.remove_listener("request", on_request)
            p.remove_listener("response", on_response)
        req = got["req"]
        base_url = req["url"].split("?")[0]
        hdr = {k: v for k, v in req["headers"].items() if k.lower() not in ("cookie", "content-length", "host")}
        items, off, total = [], 0, None
        t1 = time.monotonic()
        while True:
            res = p.evaluate("""async ([u, b, h]) => { const r = await fetch(u, {method: 'POST', body: b, headers: h,
                                credentials: 'include'}); return {s: r.status, t: await r.text()}; }""",
                             [f"{base_url}?limit={config.DRILL_PAGE}&offset={off}", req["body"], hdr])
            try:
                pay = json.loads(res["t"])["responsePayload"]
            except (ValueError, KeyError, TypeError):
                raise DrillError(f"getData returned HTTP {res['s']} without data at offset {off}.") from None
            if total is None:
                total = int(pay.get("totalResults") or 0)
                if total > max_lines:
                    raise TooManyLines(total)
            items += pay.get("items") or []
            if progress:
                progress(len(items), total)
            if not pay.get("hasMore") or not pay.get("items"):
                break
            off += len(pay["items"])
        if total is not None and len(items) != total:
            log.warning("drill paged %d items but totalResults was %d", len(items), total)
        return {"items": items, "total": total, "view": got.get("view"), "secs": round(time.monotonic() - t1, 1)}

    def fetch(self, url: str, max_lines: int = config.DRILL_MAX_LINES,
              progress: Callable[[int, int], None] | None = None) -> cf.Future:
        return self.submit(lambda: self._fetch(url, max_lines, progress))

    # ------------------------------------------------------------ cleanup
    def shutdown(self):
        def close():
            try:
                if self.ctx:
                    self.ctx.close()
                if self.pw:
                    self.pw.stop()
            finally:
                self.ctx = self.pw = self.page = None
                self.ready = False
        try:
            self.submit(close).result(timeout=30)
        except Exception:   # noqa: BLE001
            kill_ours(self.profile)
        self.q.put((None, None))

    def kill(self):
        """Hang recovery: kill only chrome.exe processes started with OUR profile dir, then start fresh next time."""
        kill_ours(self.profile)
        self.ctx = self.pw = self.page = None
        self.ready = False


def kill_ours(profile) -> list[int]:
    mark, killed = str(profile).lower(), []
    for pr in psutil.process_iter(["name", "cmdline"]):
        try:
            if (pr.info["name"] or "").lower() == "chrome.exe" and mark in " ".join(pr.info["cmdline"] or []).lower():
                pr.kill()
                killed.append(pr.pid)
        except (psutil.Error, OSError):
            pass
    return killed
```
